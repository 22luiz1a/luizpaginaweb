Funcionalidade do Modo Escuro / Modo Claro const btnTheme = document.getElementById('btn-theme'); btnTheme.addEventListener('click', () => { // Alterna a classe dark-mode no body document.body.classList.toggle('dark-mode'); // Atualiza o texto do botão de acordo com o tema ativo if (document.body.classList.contains('dark-mode')) { btnTheme.textContent = '☀️ Modo Claro'; } else { btnTheme.textContent = '🌙 Modo Escuro'; } }); // 2. Funcionalidade dos Botões de Curtida e Coração const botoesAcoes = document.querySelectorAll('.post-acoes button'); botoesAcoes.forEach((botao) => { botao.addEventListener('click', () => { // Encontra o elemento dentro do botão clicado const contadorSpan = botao.querySelector('span'); // Pega o número atual do contador e converte para número let curtidasAtuais = Number(contadorSpan.textContent); // Incrementa +1 e atualiza o texto na tela contadorSpan.textContent = curtidasAtuais + 1; }); });
